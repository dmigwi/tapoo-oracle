```js
// The redirect itself lives in the head (observablehq.config.js), because nothing loaded as a module
// can be relied on here: this page is served at /r/<token>, where its own relative asset paths
// resolve a directory too deep and 404. By the time this runs, a report link has already left.
//
// So what remains is the page a genuinely mistyped address lands on.
display(html`<p class="report-share-error">This address does not name a report. Ask for a fresh link, or
  <a href=${location.pathname.replace(/\/r\/[A-Za-z0-9_-]+\/?$/, "/")}>start from the analyzer</a>.</p>`);
```
